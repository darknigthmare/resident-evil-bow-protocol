  function validate(value) {

      const variants = new Set();

            || ![variant.sourceEdition, variant.targetIncarnation].every(item => typeof item === 'string' && item.length > 0 && item.length <= 10000)
            || variant.admittedTargetEdition !== false || variant.targetFullBodyReviewed !== false || !Array.isArray(variant.availableStates) || variant.availableStates.length) throw Error('Édition documentaire promue hors périmètre');
        variants.add(variant.variantId);
      }
      let views = 0;
      for (const source of row.bodySources) {

              || !/^(?:sources\/(?:body-rasters|revelations-body-rasters)|motion)\/[a-zA-Z0-9_./-]+\.png$/.test(view.path)
              || view.path.split('/').some(part => !part || part === '.' || part === '..')
              || !/^[a-f0-9]{64}$/.test(view.sha256) || !Number.isSafeInteger(view.bytes) || view.bytes < 1 || view.bytes > 16 * 1024 * 1024) throw Error('Vue corporelle R29 hors périmètre');

        }
      }
      if (views !== row.bodyViews) throw Error('Compteur de vues R29 incompatible');
      const pose = row.spriteChange;
      if (pose !== null && (!pose || typeof pose.fullBodyReviewed !== 'boolean' || typeof pose.pixelRevision !== 'boolean'

          || typeof pose.sourceBodyObserved !== 'boolean' || pose.sourceBodyObserved !== row.activeSourceBodyObserved
          || !['neutral-base', 'source-bind-pose', 'source-presentation', 'unknown'].includes(pose.poseKind)
          || !['left', 'right', 'front', 'unknown'].includes(pose.sourceFacing)
          || !['combatant-neutral-pose', 'context-encounter'].includes(pose.scope)
          || !Array.isArray(pose.knownAppearanceGaps) || pose.knownAppearanceGaps.some(item => typeof item !== 'string' || item.length > 10000)
          || !Array.isArray(pose.availableStates) || pose.availableStates.length > 1 || pose.availableStates.some(item => item !== 'idle')
          || pose.knownAppearanceGaps.length > 0 && pose.availableStates.length > 0
          || pose.availableStates.includes('idle') && (!pose.fullBodyReviewed || !pose.sourceBodyObserved || pose.poseKind !== 'neutral-base' || !['left', 'right'].includes(pose.sourceFacing) || pose.scope !== 'combatant-neutral-pose'))) throw Error('Portée de pose R29 incompatible');

      try { while (true) { const part = await reader.read(); if (part.done) break; length += part.value.length; if (length > 512 * 1024) throw Error('Index R29 trop volumineux'); chunks.push(part.value); } }

      entries.clear(); for (const [id, row] of next) entries.set(id, row); state.failed = false;
    } catch (error) { if (token !== ticket || error.name === 'AbortError') return; state.failed = true; entries.clear(); }
    state.loaded = true; sync();
  }
